```python
class CrossTaskAttention(nn.Module):
    """跨任务注意力：让选股任务关注择时信息"""
    
    def __init__(self, d_model):
        super().__init__()
        self.attention = nn.MultiheadAttention(d_model, num_heads=4)
    
    def forward(self, stock_features, market_state):
        """
        Args:
            stock_features: [batch, n_stocks, d_model]
            market_state: [batch, d_model]
        """
        # 将市场状态作为query，股票特征作为key和value
        market_query = market_state.unsqueeze(1)  # [batch, 1, d_model]
        attended, _ = self.attention(
            market_query.transpose(0, 1),
            stock_features.transpose(0, 1),
            stock_features.transpose(0, 1)
        )
        return attended.transpose(0, 1)
```

### 1.3 因子融合机制设计

**方案A：注意力加权融合（推荐）**
```python
class FactorFusion(nn.Module):
    """因子融合模块：自适应学习因子权重"""
    
    def __init__(self, n_factors, d_model):
        super().__init__()
        # 因子重要性学习
        self.factor_importance = nn.Sequential(
            nn.Linear(n_factors, n_factors),
            nn.Tanh(),
            nn.Linear(n_factors, n_factors),
            nn.Softmax(dim=-1)
        )
        # 因子嵌入
        self.factor_embed = nn.Linear(n_factors, d_model)
    
    def forward(self, factors):
        """
        Args:
            factors: [batch, seq_len, n_factors]
        """
        # 计算因子权重
        weights = self.factor_importance(factors)  # [batch, seq_len, n_factors]
        
        # 加权因子
        weighted_factors = factors * weights
        
        # 嵌入
        embedded = self.factor_embed(weighted_factors)
        
        return embedded, weights  # 返回权重用于可解释性分析
```

**方案B：分组融合**
```python
# 将因子分组：估值、盈利、成长、质量、动量、波动
factor_groups = {
    'valuation': ['pe_ttm', 'pb', 'ps_ttm'],
    'profitability': ['roe', 'roa', 'net_margin'],
    'growth': ['revenue_growth', 'profit_growth'],
    'quality': ['debt_to_asset', 'current_ratio'],
    'momentum': ['return_1m', 'return_3m', 'return_6m'],
    'volatility': ['volatility_20d', 'beta_60d']
}

# 每组独立编码后再融合
class GroupedFactorFusion(nn.Module):
    def __init__(self, factor_groups, d_model):
        super().__init__()
        self.group_encoders = nn.ModuleDict({
            name: nn.Linear(len(factors), d_model // len(factor_groups))
            for name, factors in factor_groups.items()
        })
        self.fusion = nn.Linear(d_model, d_model)
    
    def forward(self, factor_dict):
        # 每组因子独立编码
        group_embeddings = []
        for name, encoder in self.group_encoders.items():
            group_factors = factor_dict[name]
            group_embed = encoder(group_factors)
            group_embeddings.append(group_embed)
        
        # 拼接融合
        concatenated = torch.cat(group_embeddings, dim=-1)
        fused = self.fusion(concatenated)
        return fused
```

### 2.3 回测引擎增强

**添加滑点模型：**
```python
def calculate_slippage(self, order_size, avg_volume, volatility):
    """
    更真实的滑点模型
    
    Args:
        order_size: 订单金额
        avg_volume: 平均成交额
        volatility: 波动率
    """
    # 市场冲击成本 = f(订单占比, 波动率)
    volume_ratio = order_size / avg_volume
    
    # 平方根模型
    impact = 0.1 * volatility * np.sqrt(volume_ratio)
    
    # 固定滑点
    fixed_slippage = 0.0005
    
    return fixed_slippage + impact
```

**添加停牌处理：**
```python
def handle_suspension(self, portfolio, suspended_stocks):
    """
    处理停牌股票
    
    Args:
        portfolio: 目标组合
        suspended_stocks: 停牌股票列表
    """
    # 从目标组合中移除停牌股票
    adjusted_portfolio = {
        code: weight 
        for code, weight in portfolio.items()
        if code not in suspended_stocks
    }
    
    # 重新归一化权重
    total_weight = sum(adjusted_portfolio.values())
    if total_weight > 0:
        adjusted_portfolio = {
            code: weight / total_weight
            for code, weight in adjusted_portfolio.items()
        }
    
    return adjusted_portfolio
```

```python
from concurrent.futures import ThreadPoolExecutor, as_completed

def parallel_factor_calculation(stock_codes, calc_func):
    """并行计算因子"""
    results = {}
    
    with ThreadPoolExecutor(max_workers=8) as executor:
        futures = {
            executor.submit(calc_func, code): code
            for code in stock_codes
        }
        
        for future in tqdm(as_completed(futures), total=len(futures)):
            code = futures[future]
            try:
                results[code] = future.result()
            except Exception as e:
                print(f"计算 {code} 失败: {e}")
    
    return results
```

```python
def factor_contribution_analysis(model, factors, factor_names):
    """分析每个因子的贡献度"""
    model.eval()
    baseline_score = model(factors).item()
    
    contributions = {}
    for i, name in enumerate(factor_names):
        # 将第i个因子置零
        perturbed_factors = factors.clone()
        perturbed_factors[:, :, i] = 0
        
        perturbed_score = model(perturbed_factors).item()
        contribution = baseline_score - perturbed_score
        contributions[name] = contribution
    
    # 排序
    sorted_contrib = sorted(
        contributions.items(),
        key=lambda x: abs(x[1]),
        reverse=True
    )
    
    return sorted_contrib
```

```python
def construct_industry_neutral_portfolio(stock_scores, stock_industry, 
                                        benchmark_weights):
    """构建行业中性组合"""
    # 每个行业内选股
    portfolio = {}
    
    for industry in stock_industry.unique():
        # 该行业的股票
        industry_mask = stock_industry == industry
        industry_scores = stock_scores[industry_mask]
        industry_stocks = stock_industry.index[industry_mask]
        
        # 基准权重
        benchmark_weight = benchmark_weights[industry]
        
        # 选取该行业top股票
        n_select = max(1, int(len(industry_stocks) * 0.2))
        top_indices = np.argsort(industry_scores)[-n_select:]
        
        # 分配权重（等权）
        weight_per_stock = benchmark_weight / n_select
        for idx in top_indices:
            portfolio[industry_stocks[idx]] = weight_per_stock
    
    return portfolio
```

### 5.2 风险预算

```python
def risk_parity_weights(expected_returns, cov_matrix):
    """风险平价权重优化"""
    from scipy.optimize import minimize
    
    n_assets = len(expected_returns)
    
    def risk_budget_objective(weights, cov_matrix):
        # 计算每个资产的风险贡献
        portfolio_vol = np.sqrt(weights @ cov_matrix @ weights)
        marginal_contrib = cov_matrix @ weights
        risk_contrib = weights * marginal_contrib / portfolio_vol
        
        # 目标：使所有资产风险贡献相等
        target = portfolio_vol / n_assets
        return np.sum((risk_contrib - target) ** 2)
    
    # 约束：权重和为1，权重非负
    constraints = [
        {'type': 'eq', 'fun': lambda w: np.sum(w) - 1}
    ]
    bounds = [(0, 0.1)] * n_assets  # 单只股票最大10%
    
    # 初始权重
    w0 = np.ones(n_assets) / n_assets
    
    # 优化
    result = minimize(
        risk_budget_objective,
        w0,
        args=(cov_matrix,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )
    
    return result.x
```

### 6.1 模型融合

```python
class EnsembleModel:
    """模型集成：提高稳定性"""
    
    def __init__(self, models):
        self.models = models
    
    def predict(self, market_factors, stock_factors):
        """集成预测"""
        regime_preds = []
        ranking_preds = []
        
        for model in self.models:
            model.eval()
            with torch.no_grad():
                regime_logits, ranking_scores = model(
                    market_factors, stock_factors
                )
                regime_preds.append(regime_logits)
                ranking_preds.append(ranking_scores)
        
        # 平均集成
        regime_ensemble = torch.stack(regime_preds).mean(dim=0)
        ranking_ensemble = torch.stack(ranking_preds).mean(dim=0)
        
        return regime_ensemble, ranking_ensemble
```
